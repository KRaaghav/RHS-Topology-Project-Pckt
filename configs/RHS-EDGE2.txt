license udi pid CISCO2911/K9 sn FTX1524AG5Q-

! ---- SSH setup. Needs a domain name, and "crypto key generate rsa" typed once on the device
ip ssh version 2
no ip domain-lookup
ip domain-name rhs.lab

!
! ---- LLDP so "show lldp neighbors" shows what is plugged in where
lldp run

spanning-tree mode pvst

!
! ---- Loopback: always up, used as the OSPF router ID
interface Loopback0
 description Router ID
 ip address 10.10.255.4 255.255.255.255
!
! ---- Inside ports to the core switches: NAT inside, OSPF with faster timers
interface GigabitEthernet0/0
 description MDF - to CORE-L3-1 Gi1/0/4
 ip address 10.10.0.13 255.255.255.252
 ip ospf network point-to-point
 ip ospf hello-interval 2
 ip ospf dead-interval 8
 ip nat inside
 duplex auto
 speed auto
 ipv6 address 2001:DB8:10:F3::1/64
!
interface GigabitEthernet0/1
 description MDF - to CORE-L3-2 Gi1/0/4
 ip address 10.10.0.17 255.255.255.252
 ip ospf network point-to-point
 ip ospf hello-interval 2
 ip ospf dead-interval 8
 ip nat inside
 duplex auto
 speed auto
 ipv6 address 2001:DB8:10:F4::1/64
!
! ---- Unused port
interface GigabitEthernet0/2
 no ip address
 duplex auto
 speed auto
 shutdown
!
! ---- WAN port to the ISP: NAT outside + the inbound ACL
interface GigabitEthernet0/0/0
 description WAN fiber 2 to ISP
 ip address 203.0.113.6 255.255.255.252
 ip access-group OUTSIDE-IN in
 ip nat outside
 ipv6 address 2001:DB8:FFFE::2/64
 no cdp enable
!
! ---- Unused port
interface GigabitEthernet0/1/0
 no ip address
 shutdown
!
! ---- VLAN 1 is not used
interface Vlan1
 no ip address
 shutdown
!
! ---- OSPF, single area 0. Only the links to other routers send hellos, everything else is passive
router ospf 1
 router-id 10.10.255.4
 log-adjacency-changes
 passive-interface Loopback0
 network 10.10.0.12 0.0.0.3 area 0
 network 10.10.0.16 0.0.0.3 area 0
 network 10.10.255.4 0.0.0.0 area 0
 default-information originate metric 100
!
! ---- NAT: PAT for all inside addresses out the WAN port, plus a static NAT for the school web server
ip nat inside source list 1 interface GigabitEthernet0/0/0 overload
ip nat inside source static 10.10.70.10 203.0.113.10
ip classless
! ---- Default route to the ISP
ip route 0.0.0.0 0.0.0.0 203.0.113.5
!
! ---- IPv6 static routes. The ones ending in 5 are floating backups (higher administrative distance)
ipv6 route ::/0 2001:DB8:FFFE::1
ipv6 route 2001:DB8:10::/48 2001:DB8:10:F3::2
ipv6 route 2001:DB8:10::/48 2001:DB8:10:F4::2 5
!
! ---- Standard ACL 1: which inside addresses get translated by NAT
access-list 1 permit 10.10.0.0 0.0.255.255
! ---- Standard ACL 10: only the Network IT VLAN may SSH in (used by access-class on the vty lines)
access-list 10 permit 10.10.99.0 0.0.0.255
! ---- Traffic coming in from the internet: replies and the public web server only
ip access-list extended OUTSIDE-IN
 permit tcp any host 203.0.113.10 eq 80
 permit tcp any host 203.0.113.10 eq 443
 permit tcp any any established
 permit udp any eq 53 any
 permit icmp any any echo-reply
 permit icmp any any unreachable
 deny ip any any
!
!
! ---- Warning shown before login
banner motd ^CRHS campus network (lab design). Authorized users only - activity is logged.^C

! ---- Send logs to the syslog server SRV-LOG

logging 10.10.70.20

! ---- Read only SNMP community
snmp-server community RHS-RO RO

! ---- Console port: asks for the local username and password
line con 0
 exec-timeout 15 0

 login local
!
line aux 0
!
! ---- Remote access: SSH only, local login, limited by ACL 10
line vty 0 4
 access-class 10 in
 exec-timeout 10 0
 login local
 transport input ssh
!
! ---- Get the time from SRV-LOG so log timestamps match
ntp server 10.10.70.20
